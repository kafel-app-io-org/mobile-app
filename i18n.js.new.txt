// i18n.js (multi-language)
import i18n from 'i18next';
import { initReactI18next } from 'react-i18next';
import * as Localization from 'expo-localization';
import { I18nManager, Platform } from 'react-native';
import * as Updates from 'expo-updates';

// Translation files
import en from './languages/en.json';
import ar from './languages/ar.json';
import tr from './languages/tr.json';
import { i18nStorage } from "./utils/i18nStorage";

const LANGUAGE_STORAGE_KEY = 'user-language';

export const SUPPORTED_LANGUAGES = [
  { code: 'en', label: 'English', isRTL: false },
  { code: 'ar', label: 'العربية', isRTL: true },
  { code: 'tr', label: 'Türkçe', isRTL: false },
];

const supportedCodes = SUPPORTED_LANGUAGES.map((l) => l.code);

export const getCurrentLanguage = () => i18n.language;

// keep your existing exported variable (some screens may rely on it)
export let selectedLanguage = '';

const normalizeLanguage = (lng) => {
  if (!lng) return 'en';
  const short = String(lng).split(/[-_]/)[0].toLowerCase(); // "en-US" -> "en"
  return supportedCodes.includes(short) ? short : 'en';
};

const getDeviceLanguage = () => {
  const locales =
    typeof Localization.getLocales === 'function' ? Localization.getLocales() : [];
  const code =
    locales?.[0]?.languageCode ||
    Localization.locale?.split(/[-_]/)[0] ||
    'en';
  return normalizeLanguage(code);
};

const isRtlLanguage = (lng) =>
  SUPPORTED_LANGUAGES.find((l) => l.code === lng)?.isRTL ?? false;

const languageDetector = {
  type: 'languageDetector',
  async: true,
  detect: async (callback) => {
    try {
      const savedLang = await i18nStorage.getItem(LANGUAGE_STORAGE_KEY);
      const lang = normalizeLanguage(savedLang || getDeviceLanguage() || 'en');
      selectedLanguage = lang;
      callback(lang);
    } catch (err) {
      console.error('Language detection error:', err);
      selectedLanguage = 'en';
      callback('en');
    }
  },
  init: () => {},
  cacheUserLanguage: async (lng) => {
    try {
      await i18nStorage.setItem(LANGUAGE_STORAGE_KEY, normalizeLanguage(lng));
    } catch (err) {
      console.error('Failed to cache language:', err);
    }
  },
};

// Set language explicitly (and apply RTL/LTR if needed)
export const setLanguage = async (lng) => {
  const nextLang = normalizeLanguage(lng);

  try {
    await i18nStorage.setItem(LANGUAGE_STORAGE_KEY, nextLang);
    selectedLanguage = nextLang;

    const shouldBeRTL = isRtlLanguage(nextLang);

    // If direction changes, RN needs a reload to apply layout direction.
    if (Platform.OS !== 'web') {
      const rtlChanged = I18nManager.isRTL !== shouldBeRTL;

      if (rtlChanged) {
        I18nManager.allowRTL(true);
        I18nManager.forceRTL(shouldBeRTL);

        // change language before reload so next boot shows correct strings
        await i18n.changeLanguage(nextLang);

        try {
          await Updates.reloadAsync();
        } catch (e) {
          console.warn('RTL change may require an app restart.', e);
        }
        return;
      }
    }

    await i18n.changeLanguage(nextLang);
  } catch (err) {
    console.error('Error setting language:', err);
  }
};

// Cycle language: en -> ar -> tr -> en ...
export const switchLanguage = async () => {
  try {
    const currentStored = await i18nStorage.getItem(LANGUAGE_STORAGE_KEY);
    const current = normalizeLanguage(currentStored || i18n.language || 'en');

    const idx = supportedCodes.indexOf(current);
    const next = supportedCodes[(idx + 1) % supportedCodes.length];

    await setLanguage(next);
  } catch (err) {
    console.error('Error switching language:', err);
  }
};

i18n
  .use(languageDetector)
  .use(initReactI18next)
  .init({
    fallbackLng: 'en',
    supportedLngs: supportedCodes,
    nonExplicitSupportedLngs: true,
    load: 'languageOnly',
    resources: {
      en: { translation: en },
      ar: { translation: ar },
      tr: { translation: tr },
    },
    interpolation: { escapeValue: false },
    react: { useSuspense: false },
  });

export default i18n;
